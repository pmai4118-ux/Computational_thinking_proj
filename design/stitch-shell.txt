<script>
  const { screens, assets } = JSON.parse(document.getElementById('stitch-data').textContent);
  const byId = new Map(screens.map(screen => [screen.id, screen]));
  const frame = document.getElementById('screen');
  const picker = document.getElementById('screen-picker');
  const list = document.getElementById('screen-list');
  const legacy = {
    'cau-hoi': 'question-library', 'phong-van': 'interview-setup', 'lo-trinh': 'learning-roadmap',
    'tong-quan': 'dashboard', 'dang-nhap': 'auth-google', 'dau-vao': 'entry-assessment'
  };
  let active;
  function render() {
    const key = location.hash.slice(1);
    const screen = byId.get(legacy[key] || key) || screens[0];
    if (active === screen.id) return;
    active = screen.id;
    document.title = screen.title + ' · interviewUS';
    frame.title = screen.title;
    frame.srcdoc = screen.html.replace(/@@STITCH_ASSET_(\d+)@@/g, (_, index) => assets[Number(index)]);
    for (const button of list.children) {
      if (button.dataset.screen === active) button.setAttribute('aria-current', 'page');
      else button.removeAttribute('aria-current');
    }
  }
  function navigate(target) {
    if (!byId.has(target)) return;
    picker.open = false;
    location.hash = target;
    render();
  }
  for (const screen of screens) {
    const button = document.createElement('button');
    button.type = 'button';
    button.dataset.screen = screen.id;
    button.textContent = screen.title;
    button.addEventListener('click', () => navigate(screen.id));
    list.append(button);
  }
  window.addEventListener('hashchange', render);
  window.addEventListener('message', event => {
    if (event.source === frame.contentWindow && event.data?.type === 'stitch-demo:navigate') navigate(event.data.target);
  });
  render();
</script>
